def get_passing(scores):
    """Return names whose score is 50 or more."""
    result = []
    for name, score in scores.items():
        if score >= 50:
            result.append(name)
    return result
 
scores = {"Nour": 92, "Omar": 44, "Sara": 78}
print(get_passing(scores))
Output
['Nour', 'Sara']
