def menu():
    print("1) Add 2) Subtract 3) Multiply 4) Divide 5) Exit")
 
def main():
    while True:
        menu()
        choice = input("Choose: ")
        if choice == "5":
            break
        a = float(input("First number: "))
        b = float(input("Second number: "))
        if choice == "1": print(add(a, b))
        elif choice == "2": print(subtract(a, b))
        elif choice == "3": print(multiply(a, b))
        elif choice == "4": print(divide(a, b))
    print("Bye!")
 
if __name__ == "__main__": # run when executed
    main()
Structure 🏗️
🖥️ menu() — show options
🔄 main() — loop + dispatch
🚀 __main__ — entry point
