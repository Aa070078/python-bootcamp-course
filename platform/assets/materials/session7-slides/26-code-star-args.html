def add_all(*numbers):
    total = 0
    for n in numbers: # numbers is a tuple
        total += n
    return total
 
print(add_all(1, 2, 3)) # 6
print(add_all(4, 5, 6, 7)) # 22
 
def show_info(**info):
    for key, value in info.items():
        print(f"{key}: {value}")
 
show_info(name="Nour", level=3)
