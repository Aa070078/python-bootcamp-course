profile.py
def profile(name, age, city):
    print(f"{name} | {age} | {city}")
 
# same call, different order — keywords do the matching
profile(age=19, city="Cairo", name="Nour")
 
# mixing: positional first, then keywords
profile("Omar", city="Dubai", age=21)
Output
Nour | 19 | Cairo
Omar | 21 | Dubai
