def average(scores):
    """Return the mean of a list of numbers."""
    return sum(scores) / len(scores)
 
def grade_report(students):
    """Print average + pass/fail for each student."""
    for name, scores in students.items():
        avg = average(scores)
        status = "PASS" if avg >= 50 else "FAIL"
        print(f"{name}: {avg:.1f} -> {status}")
 
report = {"Nour": [90, 88, 95], "Omar": [40, 55, 48]}
grade_report(report)

Omar: 47.7 -> FAIL
