def cel_to_far(celsius):
    """Convert Celsius to Fahrenheit.
     Args: celsius (float): temperature in °C.
     Returns: temperature in °F.
    """
    return celsius * 9 / 5 + 32
 
help(cel_to_far) # prints the docstring!
print(cel_to_far.__doc__)
