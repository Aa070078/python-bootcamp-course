def greet(name, greeting="Hello"): # default param
    """Say hi with an optional greeting.""" # docstring
    return f"{greeting}, {name}!" # return hands back a value
 
greet("Nour") # positional
greet(name="Nour", greeting="Salam") # keyword
 
def flex(*args, **kwargs): # tuple + dict
    return args, kwargs
 
double = lambda x: x * 2 # lambda
 
if __name__ == "__main__": # entry point
    print(greet("Sara"))
Golden rules ⭐
✔️ One job per function
✔️ Use return, not print
✔️ snake_case names = verbs
✔️ Defaults go last
✔️ Local beats global
✔️ Small > clever
