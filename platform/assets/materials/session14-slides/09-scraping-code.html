import requests
from bs4 import BeautifulSoup
url = "https://quotes.toscrape.com/"
resp = requests.get(url) # fetch the page
soup = BeautifulSoup(resp.text, "html.parser") # parse it
for quote in soup.select(".quote"):
    print(quote.select_one(".text").text)
WHAT HAPPENS?
1️⃣ `requests` downloads the page's HTML.
2️⃣ `BeautifulSoup` turns it into a tree.
3️⃣ `.select(".quote")` grabs every quote block.
4️⃣ we print the text of each one.
