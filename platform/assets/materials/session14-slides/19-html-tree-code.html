BeautifulSoup يحوّل نص HTML إلى شجرة نتنقل فيها مثل القائمة المتداخلة — توقّع الناتج

from bs4 import BeautifulSoup
html = "<html><body><h1>Hello</h1><p>Bootcamp</p></body></html>"
soup = BeautifulSoup(html, "html.parser")
print(soup.body.h1.text) # walk body → h1
print(soup.body.p.text) # walk body → p
print(soup.body.get_text(", ")) # all text joined
