import csv, time, requests
from bs4 import BeautifulSoup
rows = []
for page in range(1, 4):
    soup = BeautifulSoup(requests.get(
        f("https://quotes.toscrape.com/page/{page}/"), timeout=10).text,
        "html.parser")
    for q in soup.select(".quote"):
        rows.append([q.select_one(".text").get_text().strip(),
                 q.select_one(".author").get_text()])
    time.sleep(2)
with open("quotes.csv", "w", newline="", encoding="utf-8") as f:
    w = csv.writer(f); w.writerow(["text", "author"]); w.writerows(rows)
print(f("Saved {len(rows)} quotes"))
