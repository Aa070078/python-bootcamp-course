import requests
from bs4 import BeautifulSoup
url = "https://quotes.toscrape.com/"
resp = requests.get(url, timeout=10)
soup = BeautifulSoup(resp.text, "html.parser")
first = soup.select_one(".quote") # one quote block
print(first.select_one(".text").get_text())
print("--", first.select_one(".author").get_text())
