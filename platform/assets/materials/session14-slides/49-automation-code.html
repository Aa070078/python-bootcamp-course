نموذج Playwright: افتح صفحة، املأ نموذجًا، اضغط زرًا، وانتظر النتيجة
automate.py
from playwright.sync_api import sync_playwright
with sync_playwright() as p:
    browser = p.chromium.launch(headless=True)
    page = browser.new_page()
    page.goto("https://example.com/login") # open
    page.fill("#user", "bootcamper") # fill
    page.fill("#pass", "hunter2")
    page.click("button[type=submit]") # click
    page.wait_for_selector(".dashboard") # wait
    print(page.title())
    browser.close()
