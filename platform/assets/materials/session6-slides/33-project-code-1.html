# Contact Manager — Part 1: data + add
contacts = []   # each contact is a dict

def add_contact():
    name = input("Name: ").strip().title()
    phone = input("Phone: ").strip()
    tags = {t.strip() for t in input("Tags: ").split(",")}
    contacts.append({
        "name": name,
        "phone": phone,
        "tags": tags
    })
    print(f"Added {name}!")

def show_tags():
    all_tags = set()
    for c in contacts:
        all_tags |= c["tags"]   # union
    print("Unique tags:", sorted(all_tags))
