Search + remove + menu loop
contact_manager.py 
def search():
    word = input("Search name: ").strip().lower()
    for c in contacts:
        if word in c["name"].lower():
            print(c["name"], "|", c["phone"],
                  "| tags:", sorted(c["tags"]))

def remove():
    name = input("Remove: ").strip().lower()
    for i, c in enumerate(contacts):
        if c["name"].lower() == name:
            contacts.pop(i)
            print(f"Removed {c['name']}.")
            return
    print("Not found.")
menu loop 
while True:
    print("1 add | 2 search | 3 remove | 4 tags | 5 quit")
    choice = input("> ")
    if choice == "1":
        add_contact()
    elif choice == "2":
        search()
    elif choice == "3":
        remove()
    elif choice == "4":
        show_tags()
    elif choice == "5":
        break
    else:
        print("Invalid choice.")
print("Bye!")
🔑 The menu is a dispatcher: it reads one number and calls exactly one function. break exits the loop → "Bye!"
