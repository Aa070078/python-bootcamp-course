sets.py 
# SETS — unique, unordered, no duplicates
colors = {"red", "green", "blue"}
colors.add("orange")          # add one
colors.discard("red")         # safe remove
colors.remove("blue")         # error if missing
"green" in colors             # True

a = {1, 2, 3}
b = {3, 4, 5}
a | b       # union         -> {1,2,3,4,5}
a & b       # intersection  -> {3}
a - b       # difference    -> {1,2}
a ^ b       # symmetric     -> {1,2,4,5}

unique = list(set([1, 1, 2, 3, 3]))  # dedupe
dicts.py 
# DICTS — key -> value pairs
phonebook = {"Sara": "050-1234", "Omar": "077-5678"}
phonebook["Lina"] = "055-9999"   # add / update
del phonebook["Omar"]            # remove
phonebook.get("Sara")            # "050-1234"
phonebook.get("Ali", "none")     # default value

for name, num in phonebook.items():
    print(name, num)

counts = {}
counts["a"] = counts.get("a", 0) + 1   # counting
# COMBINING — list of dicts, set values
team = [{"name": "Sara", "tags": {"work", "friend"}},
         {"name": "Omar", "tags": {"work", "gym"}}]
all_tags = set()
for m in team: all_tags |= m["tags"]
print(sorted(all_tags)) # ['friend','gym','work']
