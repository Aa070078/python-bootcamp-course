pass-vs-continue.py 
# pass vs continue
for i in range(5):
    if i == 3:
        pass
    print(i)
# Output: 0 1 2 3 4

for i in range(5):
    if i == 3:
        continue
    print(i)
# Output: 0 1 2 4
pass = placeholder (loop continues normally)
continue = skip this iteration
