# Count vowels
word = "Hello World"
count = 0
for char in word:
    if char.lower() in "aeiou":
        count += 1
print(f"Vowels: {count}")
password-check.py 
# Password validation
password = "Py!2026"
has_upper = False
has_digit = False
has_special = False

for char in password:
    if char.isupper():
        has_upper = True
    elif char.isdigit():
        has_digit = True
    elif char in "!@#$":
        has_special = True

print(f"Upper:{has_upper} Digit:{has_digit} Special:{has_special}")
