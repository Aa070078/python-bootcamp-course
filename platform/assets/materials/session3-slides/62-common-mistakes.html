Forgetting to update
WRONG
while x < 10:
    print(x)
    # forgot x += 1!
CORRECT
while x < 10:
    print(x)
    x += 1
Off-by-one error
WRONG
for i in range(1, 10):
    print(i)
# Prints 1-9, not 1-10!
CORRECT
for i in range(1, 11):
    print(i)
# range is exclusive!
Modifying list while iterating
WRONG
nums = [1,2,3,4,5]
for n in nums:
    if n % 2 == 0:
        nums.remove(n)
CORRECT
nums = [1,2,3,4,5]
evens = [n for n in nums
          if n % 2 != 0]
Using = instead of ==
WRONG
for i in range(5):
    if i = 3:  # ERROR!
        print("found")
CORRECT
for i in range(5):
    if i == 3:  # correct
        print("found")
