# Basic list iteration
students = ["Ali", "Sara", "Omar", "Lina"]

for student in students:
    print(f"Welcome, {student}!")

# Sum of list
scores = [85, 92, 78, 95, 88]
total = 0
for score in scores:
    total += score
print(f"Average: {total / len(scores)}")
