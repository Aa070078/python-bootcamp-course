import random

secret = random.randint(1, 50)
attempts = 0
max_attempts = 7

print("Guess the number (1-50)!")

while attempts < max_attempts:
    guess = int(input("Your guess: "))
    attempts += 1

    if guess == secret:
        print(f"Correct in {attempts} tries!")
        break
    elif guess < secret:
        print("Too low!")
    else:
        print("Too high!")
else:
    print(f"Game over! The number was {secret}")
