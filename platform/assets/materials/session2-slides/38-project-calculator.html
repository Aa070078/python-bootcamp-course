n1 = float(input("Number 1: "))
op = input("Operator (+ - * /): ")
n2 = float(input("Number 2: "))
if op == "+":
print(n1 + n2)
elif op == "-":
print(n1 - n2)
elif op == "*":
print(n1 * n2)
elif op == "/":
print(n1 / n2)
else:
print("Invalid operator")
