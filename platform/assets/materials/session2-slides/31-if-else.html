// FAMILY 02 OF 03
if / else
Two paths — one for True, one for False.
// IF ALONE — ONE PATH
if_only.py
if rainy:
    print("Take umbrella")
# Not rainy? → nothing happens
# No else needed
// IF / ELSE — TWO PATHS
if_else.py
if rainy:
    print("Take umbrella")
else:
    print("Wear sunglasses")
// REAL EXAMPLE — AGE CHECK
age_check.py
age = 15
if age >= 18:
    print("Adult")
else:
    print("Minor")
