import requests

# GET with params
resp = requests.get(url, params={"key": "value"})
resp.raise_for_status()
data = resp.json()

# POST with a body
resp = requests.post(url, json={"title": "Hi"})

# friendly error handling
if resp.status_code != 200:
    print("Oops:", resp.status_code)

try: requests.get(url)
except requests.exceptions.RequestException: print("offline")
