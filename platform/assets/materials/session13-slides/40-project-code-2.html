Convert & protect
Add the user input, the math, and a graceful error path — all in one small program.
import requests

def convert(amount, from_cur, to_cur):
    resp = requests.get(
        "https://api.frankfurter.app/latest",
        params={"from": from_cur, "to": to_cur},
    )
    if resp.status_code != 200:
        print("Invalid currency or server error.")
        return None
    rate = resp.json()["rates"][to_cur]
    return round(amount * rate, 2)

amount = float(input("Amount: "))
result = convert(amount, "USD", "EUR")
print(f("{amount} USD = {result} EUR"))
DECISIONS
📥 input() — user types the amount
➗ amount * rate — the math
🛡️ status check — friendly message, no crash
🔁 round(, 2) — clean money output
