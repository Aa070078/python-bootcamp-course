import json
 
notes = [
    {"title": "Groceries", "done": False},
    {"title": "Study files", "done": True},
]
 
with open("notes.json", "w") as f:
    json.dump(notes, f, indent=2)
 
with open("notes.json") as f:
    data = json.load(f)
 
print(data[0]["title"])
