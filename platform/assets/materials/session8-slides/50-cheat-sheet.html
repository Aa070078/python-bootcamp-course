open("f.txt")  # read
open("f.txt", "w")  # wipe + write
open("f.txt", "a")  # append
📖 READ
f.read()  # all → str
f.readline()  # one line
f.readlines()  # list of lines
✍️ WRITE
f.write("hi\n")
f.writelines(lines)
🛡️ SAFE WAY
with open("f.txt", "r") as f:
    text = f.read()
📦 JSON
import json
json.dump(data, f, indent=2) # → file
json.load(f) # ← file
json.dumps(data) # → string
json.loads(text) # ← string
🛟 FAIL-SAFE LOAD
try:
    with open("f.json") as f:
        data = json.load(f)
except FileNotFoundError:
    data = []
⚠️ COMMON ERRORS
FileNotFoundError # path wrong
JSONDecodeError # file corrupt
