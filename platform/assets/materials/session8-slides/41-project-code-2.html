def main():
    notes = load_notes()
    while True:
        print("\n1) Add 2) View 3) Delete 4) Exit")
        choice = input("> ")
        if choice == "1":
            title = input("Note: ")
            notes.append({"note": title})
            save_notes(notes)
        elif choice == "2":
            for i, n in enumerate(notes, 1):
                print(i, "-", n["note"])
        elif choice == "3":
            i = int(input("Number to delete: ")) - 1
            if 0 <= i < len(notes):
                notes.pop(i)
                save_notes(notes)
        else:
            break
 
main()
Every action edits the list then saves.
1 → append + save
2 → enumerate + print
3 → pop by index + save
