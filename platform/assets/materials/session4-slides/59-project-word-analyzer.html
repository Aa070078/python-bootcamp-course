text = "Hello World 123"

chars  = len(text)
words  = len(text.split())
spaces = text.count(" ")
vowels = sum(1 for c in text.lower()
         if c in "aeiou")
digits = sum(1 for c in text
         if c.isdigit())

print(f"Characters: {chars}")   # 14
print(f"Words: {words}")        # 3
print(f"Spaces: {spaces}")      # 2
print(f"Vowels: {vowels}")      # 3
print(fD Digits: {digits}")     # 3
