word = "Python"

print(word[0:3])    # Pyt    (index 0, 1, 2)
print(word[::3])     # Pyt    (same! start defaults to 0)
print(word[2:])      # thon   (from index 2 to end)
print(word[1:4])    # yth    (index 1, 2, 3)
print(word[-3:])     # hon    (last 3 characters)
print(word[::-2])    # Pto    (every 2nd, reversed direction)
[:3] = from start to index 2
[2:] = from index 2 to end
[-3:] = last 3 characters
[::-1] = reverse everything
