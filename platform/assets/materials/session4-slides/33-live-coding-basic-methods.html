# Real-world: cleaning user input
user_input = "  PyThOn  "

# Step 1: Remove spaces
cleaned = user_input.strip()
print(cleaned)     # "PyThON"

# Step 2: Convert to lowercase
cleaned = cleaned.lower()
print(cleaned)     # "python"

# Step 3: Capitalize for display
display = cleaned.capitalize()
print(display)     # "Python"

# Step 4: Validate
if cleaned == "python":
    print("Welcome, Python learner!")
