# Basic iteration
message = "Hello"
for letter in message:
    print(letter, end=" ")
# Output: H e l l o

# With index using enumerate
for i, char in enumerate(message):
    print(f"{i}: {char}")

# Practical: count vowels
text = "Programming"
count = 0
for char in text.lower():
    if char in "aeiou":
        count += 1
print(f"Vowels: {count}")  # 3
