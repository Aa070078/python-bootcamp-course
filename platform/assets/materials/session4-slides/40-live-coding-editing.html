# Real-world: build a URL slug
title = "  My First Python Project!  "

# Step 1: Strip spaces
slug = title.strip()

# Step 2: Lowercase
slug = slug.lower()

# Step 3: Replace spaces with hyphens
slug = slug.replace(" ", "-")

# Step 4: Remove special characters
for char in "!@#$%^&*()":
    slug = slug.replace(char, "")

print(slug)  # "my-first-python-project"
