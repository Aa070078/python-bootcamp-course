F-STRINGS — DEEP DIVE
سلاسل f — شرح مفصل
fstring_advanced.py 
name = "Ahmed"
age = 20
gpa = 3.85

# Basic
print(f"Name: {name}")

# Expressions inside f-string
print(f"Next year: {age + 1}")  # 21

# Format numbers
print(f"GPA: {gpa:.2f}")  # 3.85

# Conditional in f-string
status = "Pass" if gpa >= 2.0 else "Fail"
print(f"Status: {status}")

# Align and pad
for i in range(1, 4):
    print(f"Item {i:<3} | {"*" * i}")
