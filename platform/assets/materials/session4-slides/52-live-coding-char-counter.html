char_frequency.py 
text = "hello world"
freq = {}

for char in text:
    if char in freq:
        freq[char] += 1
    else:
        freq[char] = 1

for char in sorted(freq):
    bar = "█" * freq[char]
    print(f"  '{char}': {bar} ({freq[char]})")

# Output:

#   'r': █ (1)
