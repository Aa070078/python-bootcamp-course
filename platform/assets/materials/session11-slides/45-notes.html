class Dog(Animal): # 1. blueprint (inherits from Animal)
    def __init__(self, name): # 2. constructor
        super().__init__(name) # 3. run the parent
        self.name = name # 4. attribute
        self._secret = 0 # 5. "private" (underscore)
 
    def speak(self): # 6. method (behavior)
        return "Woof!" # 7. override the parent's
 
    def __str__(self): # 8. nice printing
        return f"Dog({self.name})"
 
rex = Dog("Rex") # 9. create an object
print(rex.speak())      # 10. use it
Quick rules 📏
🐍 PascalCase for classes
🔤 snake_case for methods/vars
👮 Every method takes self
🧒 Child calls super().__init__
🔑 Change state via methods
