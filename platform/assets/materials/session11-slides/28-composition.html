composition.py
class Engine:
    def __init__(self, power):
        self.power = power
 
class Car:
    def __init__(self, brand):
        self.brand = brand
        self.engine = Engine(150)
 
car = Car("Tesla")
print(car.engine.power) # 150
Inheritance = IS-A (Dog is an Animal). Composition = HAS-A (Car has an Engine). Both are tools — use what fits.
