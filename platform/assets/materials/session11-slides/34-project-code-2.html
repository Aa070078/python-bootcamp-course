from bank_1 import SavingsAccount
 
class Bank:
    def __init__(self, name):
        self.name = name
        self.accounts = [] # composition
 
    def open_account(self, owner, balance=0):
        number = len(self.accounts) + 1001
        acc = SavingsAccount(number, owner, balance)
        self.accounts.append(acc)
        return acc
 
    def find(self, number):
        for acc in self.accounts:
            if acc.number == number:
                return acc
        return None
 
 
bank = Bank("PyBank")
sara = bank.open_account("Sara", 100)
omar = bank.open_account("Omar", 50)
 
sara.deposit(200)
omar.withdraw(20)
omar.apply_interest()
 
print(f"{bank.name} — {len(bank.accounts)} accounts")
for acc in bank.accounts:
    print(f" #{acc.number} {acc.owner}: ${acc.balance}")
