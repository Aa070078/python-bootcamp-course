methods_1.py
class Student:
    def __init__(self, name):
        self.name = name
        self.grades = []
 
    def add_grade(self, g):
        self.grades.append(g)
 
    def average(self):
        return sum(self.grades) / len(self.grades)
 
s = Student("Omar")
s.add_grade(90)
s.add_grade(80)
print(s.average()) # 85.0

class Student:
    def __init__(self, name, age):
        self.name = name
        self.age = age
 
    def __str__(self):
        return f"Student({self.name}, {self.age})"
 
s = Student("Lina", 17)
print(s) # Student(Lina, 17)
