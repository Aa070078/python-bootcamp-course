inheritance.py
class Animal:
    def __init__(self, name):
        self.name = name
 
class Dog(Animal): # Dog IS-A Animal
    pass
 
rex = Dog("Rex")
print(rex.name)                # Rex
print(isinstance(rex, Animal)) # True
🎁 Dog automatically gets everything Animal has — the __init__, name, and any methods — for free.
Write it once. Reuse it everywhere.
