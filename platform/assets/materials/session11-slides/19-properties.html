CONTROL // @property
Properties & "private" attributes
الشرطة السفلية تعني "داخلي"، و@property تمنح تحكمًا آمنًا
properties.py
class Account:
    def __init__(self, balance):
        self._balance = balance
 
    @property
    def balance(self):
        return self._balance
 
    @balance.setter
    def balance(self, value):
        if value < 0:
            raise ValueError("Negative balance not allowed")
        self._balance = value
 
acc = Account(100)
acc.balance = 250  # uses the setter
print(acc.balance) # 250 — uses the getter
Why bother? 🤷
✅ Read looks like a normal attribute: acc.balance
✅ Write runs validation first
⚠️ acc.balance = -5 → ValueError!
🔒 Internal _balance stays hidden
Same code to callers, extra safety under the hood.
