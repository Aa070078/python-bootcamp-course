class Course:
    def __init__(self, title, credits):
        self.title = title
        self.credits = credits
 
class Student:
    def __init__(self, name, age):
        self.name = name
        self.age = age
        self.courses = []
 
    def enroll(self, course):
        self.courses.append(course)
 
class School:
    def __init__(self, name):
        self.name = name
        self.students = []
 
    def add_student(self, student):
        self.students.append(student)
 
 
school = School("Neon High")
course = Course("Python", 3)
 
s1 = Student("Lina", 17)
s1.enroll(course)
school.add_student(s1)
 
for s in school.students:
    print(f"{s.name}, {s.age}")
    for c in s.courses:
        print(f" → {c.title} ({c.credits} cr)")
