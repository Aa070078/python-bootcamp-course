tasks = ["code", "eat", "sleep", "repeat"]

# remove(x) — delete by VALUE
tasks.remove("sleep")

# pop() — remove & return LAST
last = tasks.pop()
print(last)  # "repeat"

# pop(i) — remove & return at index
first = tasks.pop(0)

# del — delete by INDEX (no return)
del tasks[0]

print(tasks)  # []
remove(x)
Removes by value. Errors if not found.
pop(i)
Removes AND returns the value. Empty () = last item.
del list[i]
Deletes by index. Returns nothing.
