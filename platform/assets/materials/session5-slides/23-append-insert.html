cart = ["milk", "bread"]

# append(x) — add to the END
cart.append("eggs")
print(cart)
# ["milk", "bread", "eggs"]

# insert(i, x) — add at index i
cart.insert(1, "juice")
print(cart)
# ["milk", "juice", "bread", "eggs"]
