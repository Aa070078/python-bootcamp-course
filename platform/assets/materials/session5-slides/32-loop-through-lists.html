students = ["Ahmed", "Omar", "Sara", "Nour"]

# Loop over values
for name in students:
    print("Welcome", name)

# Loop + accumulate
scores = [80, 95, 70]
total = 0
for s in scores:
    total += s
print("Sum:", total)  # 245
