def show_average():
    total = 0
    for s in students:
        total += s[1]
    print(f"Average: {total / len(students):.2f}")

def show_top():
    best = students[0]
    for s in students:
        if s[1] > best[1]:
            best = s
    print(f"Top student: {best[0]} ({best[1]})")
