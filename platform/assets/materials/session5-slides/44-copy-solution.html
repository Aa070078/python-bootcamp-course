a = [1, 2, 3]
b = a.copy()  # real copy!

b.append(4)

print(a)  # [1, 2, 3]     ✅ safe
print(b)  # [1, 2, 3, 4]

# Slicing also copies!
c = a[:]
print(c)  # [1, 2, 3]
b = a.copy()
